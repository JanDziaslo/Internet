SELECT COUNT(*) AS liczba_zamowien
FROM Zamowienia
JOIN Stoliki ON Zamowienia.idStolika = Stoliki.id
WHERE Stoliki.numerStolika = 3 AND Stoliki.sala = 1;

Zapytanie 3

SELECT DISTINCT Potrawy.nazwa
FROM Potrawy
JOIN ZamowieniaPotrawy ON Potrawy.id = ZamowieniaPotrawy.idPotrawy
JOIN Zamowienia ON ZamowieniaPotrawy.idZamowienia = Zamowienia.id
WHERE DATE(Zamowienia.dataZlozenia) = '2024-06-09';
